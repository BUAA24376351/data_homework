        // ======================== Utilities ========================
        function parseInput(str) {
            return str.split(/[,，\s]+/)
                .map(s => s.trim())
                .filter(s => s !== '')
                .map(Number)
                .filter(n => !isNaN(n));
        }

        function clampArray(arr, maxLen = 20) {
            return arr.slice(0, maxLen);
        }

        function deepCopy(arr) {
            return arr.map(item => ({ ...item }));
        }

        function createSnapshot(array, description, focus = [], animation = '', extras = {}) {
            return {
                array: deepCopy(array),
                desc: description,
                focus,
                animation,
                extras
            };
        }

        // ======================== Step Generators ========================
        function generateBubbleSteps(arr) {
            const steps = [];
            const n = arr.length;
            const array = arr.map((v, i) => ({ value: v, state: 'default', id: i }));
            steps.push(createSnapshot(array, `初始数组: [${arr.join(', ')}]`));
            if (n <= 1) {
                array.forEach(x => x.state = 'sorted-bubble');
                steps.push(createSnapshot(array, `排序完成！`, [], '', { sortedEnd: n }));
                return steps;
            }

            for (let i = 0; i < n - 1; i++) {
                for (let j = 0; j < n - i - 1; j++) {
                    steps.push(createSnapshot(array,
                        `比较 arr[${j}]=${array[j].value} 和 arr[${j + 1}]=${array[j + 1].value}`,
                        [j, j + 1], 'compare', { sortedEnd: n - i }));

                    if (array[j].value > array[j + 1].value) {
                        steps.push(createSnapshot(array,
                            `交换 arr[${j}]=${array[j].value} 和 arr[${j + 1}]=${array[j + 1].value}`,
                            [j, j + 1], 'swap', { sortedEnd: n - i }));
                        [array[j], array[j + 1]] = [array[j + 1], array[j]];
                        steps.push(createSnapshot(array,
                            `交换完成`,
                            [j, j + 1], '', { sortedEnd: n - i }));
                    }
                    array[j].state = 'default';
                    array[j + 1].state = 'default';
                }
                array[n - i - 1].state = 'sorted-bubble';
                steps.push(createSnapshot(array,
                    `arr[${n - i - 1}] 已归位（已排序区扩大）`,
                    [n - i - 1], '', { sortedEnd: n - i }));
            }
            array.forEach(x => x.state = 'sorted-bubble');
            steps.push(createSnapshot(array, `排序完成！`, [], '', { sortedEnd: n }));
            return steps;
        }

        function generateSelectionSteps(arr) {
            const steps = [];
            const n = arr.length;
            const array = arr.map((v, i) => ({ value: v, state: 'default', id: i }));
            steps.push(createSnapshot(array, `初始数组: [${arr.join(', ')}]`));

            for (let i = 0; i < n; i++) {
                let minIdx = i;
                for (let k = 0; k < i; k++) {
                    if (array[k].state !== 'sorted-selection') array[k].state = 'sorted-selection';
                }
                array[minIdx].state = 'min-current';
                steps.push(createSnapshot(array,
                    `第 ${i + 1} 轮，当前假设最小值为 arr[${i}]=${array[i].value}`,
                    [minIdx], '', { minIdx, sortedStart: i }));

                for (let j = i + 1; j < n; j++) {
                    steps.push(createSnapshot(array,
                        `比较 arr[${minIdx}]=${array[minIdx].value} 和 arr[${j}]=${array[j].value}`,
                        [minIdx, j], 'compare', { minIdx, sortedStart: i }));

                    if (array[j].value < array[minIdx].value) {
                        const oldMinIdx = minIdx;
                        const newMinIdx = j;
                        steps.push(createSnapshot(array,
                            `发现更小的值 arr[${j}]=${array[j].value}`,
                            [oldMinIdx, newMinIdx], 'crown-transfer', { oldMinIdx, newMinIdx, sortedStart: i }));
                        array[oldMinIdx].state = 'default';
                        minIdx = newMinIdx;
                        array[minIdx].state = 'min-current';
                        steps.push(createSnapshot(array,
                            `更新最小值索引`,
                            [minIdx], '', { minIdx, sortedStart: i }));
                    }
                }

                if (minIdx !== i) {
                    steps.push(createSnapshot(array,
                        `交换 arr[${minIdx}]=${array[minIdx].value} 和 arr[${i}]=${array[i].value}`,
                        [i, minIdx], 'swap', { minIdx, sortedStart: i }));
                    [array[i], array[minIdx]] = [array[minIdx], array[i]];
                    array[i].state = 'sorted-selection';
                    array[minIdx].state = 'default';
                    steps.push(createSnapshot(array,
                        `交换完成`,
                        [i], 'arc-fly', { sortedStart: i }));
                } else {
                    array[i].state = 'sorted-selection';
                }

                steps.push(createSnapshot(array,
                    `arr[${i}] 已归位（已排序区扩大）`,
                    [i], '', { sortedStart: i + 1 }));
            }
            array.forEach(x => x.state = 'sorted-selection');
            steps.push(createSnapshot(array, `排序完成！`));
            return steps;
        }

        function generateInsertionSteps(arr) {
            const steps = [];
            const n = arr.length;
            const array = arr.map((v, i) => ({ value: v, state: 'default', id: i }));
            steps.push(createSnapshot(array, `初始数组: [${arr.join(', ')}]`));
            if (n === 0) {
                steps.push(createSnapshot(array, `排序完成！`));
                return steps;
            }

            array[0].state = 'sorted-insertion';
            steps.push(createSnapshot(array, `arr[0] 视为已排序区`, [], '', { sortedEnd: 1 }));

            for (let i = 1; i < n; i++) {
                let keyVal = array[i].value;
                let keyId = array[i].id;
                array[i] = { value: keyVal, state: 'key-picked', id: keyId };
                steps.push(createSnapshot(array,
                    `取出待插入元素 key = ${keyVal}（arr[${i}]）`,
                    [i], '', { keyIndex: i, sortedEnd: i }));

                let j = i - 1;
                while (j >= 0) {
                    steps.push(createSnapshot(array,
                        `比较 key=${keyVal} 和 arr[${j}]=${array[j].value}`,
                        [j, j + 1], 'compare', { keyIndex: j + 1, sortedEnd: i }));

                    if (array[j].value > keyVal) {
                        array[j].state = 'default';
                        array[j + 1] = array[j];
                        array[j] = { value: keyVal, state: 'key-picked', id: keyId };
                        steps.push(createSnapshot(array,
                            `arr[${j}] 后移一位到 arr[${j+1}]`,
                            [j, j + 1], 'shift', { keyIndex: j, slotIndex: j + 1, sortedEnd: i }));
                        j--;
                    } else {
                        array[j].state = 'sorted-insertion';
                        break;
                    }
                }
                if (j >= 0 && array[j].state !== 'sorted-insertion') {
                    array[j].state = 'sorted-insertion';
                }
                for (let k = 0; k <= i; k++) { if (array[k].state !== 'key-picked') array[k].state = 'sorted-insertion'; }
                array[j + 1] = { value: keyVal, state: 'key-picked', id: keyId };
                steps.push(createSnapshot(array,
                    `将 ${keyVal} 插入到 arr[${j + 1}]`,
                    [j + 1], 'drop', { keyIndex: j + 1, sortedEnd: i + 1 }));
                array[j + 1].state = 'sorted-insertion';
                steps.push(createSnapshot(array,
                    `已排序区扩大到 ${i + 1} 个元素`,
                    [], '', { sortedEnd: i + 1 }));
            }
            array.forEach(x => x.state = 'sorted-insertion');
            steps.push(createSnapshot(array, `排序完成！`));
            return steps;
        }

        function generateQuickSteps(arr) {
            const steps = [];
            const array = arr.map((v, i) => ({ value: v, state: 'default', id: i }));
            steps.push(createSnapshot(array, `初始数组: [${arr.join(', ')}]`));
            if (arr.length <= 1) {
                array.forEach(x => x.state = 'sorted-quick');
                steps.push(createSnapshot(array, `排序完成！`));
                return steps;
            }

            function partition(low, high) {
                let pivotValue = array[high].value;
                array[high].state = 'pivot';
                steps.push(createSnapshot(array,
                    `选择基准 pivot = arr[${high}] = ${pivotValue}，区间 [${low}, ${high}]`,
                    [high], '', { range: [low, high] }));

                let i = low;
                let j = high - 1;
                steps.push(createSnapshot(array,
                    `初始化左指针 i = ${low}，右指针 j = ${high - 1}`,
                    [i, j], '', { range: [low, high], i, j }));

                while (i <= j) {
                    while (i <= j && array[i].value <= pivotValue) {
                        steps.push(createSnapshot(array,
                            `左指针 i 右移到 arr[${i}] = ${array[i].value}`,
                            [i, j], '', { range: [low, high], i, j }));
                        i++;
                    }
                    if (i <= j) {
                        steps.push(createSnapshot(array,
                            `左指针 i 停在 arr[${i}] = ${array[i].value}（大于 pivot）`,
                            [i, j], '', { range: [low, high], i, j }));
                    }

                    while (i <= j && array[j].value > pivotValue) {
                        steps.push(createSnapshot(array,
                            `右指针 j 左移到 arr[${j}] = ${array[j].value}`,
                            [i, j], '', { range: [low, high], i, j }));
                        j--;
                    }
                    if (i <= j) {
                        steps.push(createSnapshot(array,
                            `右指针 j 停在 arr[${j}] = ${array[j].value}（小于等于 pivot）`,
                            [i, j], '', { range: [low, high], i, j }));
                        if (i < j) {
                            steps.push(createSnapshot(array,
                                `交换 arr[${i}]=${array[i].value} 和 arr[${j}]=${array[j].value}`,
                                [i, j], 'swap', { range: [low, high], i, j }));
                            [array[i], array[j]] = [array[j], array[i]];
                            steps.push(createSnapshot(array,
                                `交换完成`,
                                [i, j], '', { range: [low, high], i, j }));
                        }
                    }
                }

                steps.push(createSnapshot(array,
                    `左指针 i=${i} 与右指针 j=${j} 相遇（或交叉），分区结束`,
                    [i], '', { range: [low, high], i }));

                if (i !== high) {
                    steps.push(createSnapshot(array,
                        `将基准交换到 arr[${i}]`,
                        [i, high], 'swap', { range: [low, high], i }));
                    [array[i], array[high]] = [array[high], array[i]];
                    steps.push(createSnapshot(array,
                        `基准 ${array[i].value} 已归位`,
                        [i], 'pivot-place', { range: [low, high], i }));
                } else {
                    steps.push(createSnapshot(array,
                        `基准已在正确位置 arr[${i}]`,
                        [i], '', { range: [low, high], i }));
                }
                array[i].state = 'sorted-quick';
                steps.push(createSnapshot(array,
                    `基准 arr[${i}] 已归位，拆分为 [${low}, ${i - 1}] 和 [${i + 1}, ${high}]`,
                    [i], '', { range: [low, high], i, pivotPos: i }));
                return i;
            }

            function quickSort(low, high) {
                if (low < high) {
                    steps.push(createSnapshot(array,
                        `对子区间 [${low}, ${high}] 进行排序`,
                        [], '', { range: [low, high] }));
                    let p = partition(low, high);
                    quickSort(low, p - 1);
                    for (let k = low; k <= p - 1; k++) {
                        if (array[k].state !== 'sorted-quick') array[k].state = 'sorted-quick';
                    }
                    if (low <= p - 1) {
                        steps.push(createSnapshot(array,
                            `子区间 [${low}, ${p - 1}] 已有序`,
                            [], '', { range: [low, p - 1] }));
                    }
                    quickSort(p + 1, high);
                    for (let k = p + 1; k <= high; k++) {
                        if (array[k].state !== 'sorted-quick') array[k].state = 'sorted-quick';
                    }
                    if (p + 1 <= high) {
                        steps.push(createSnapshot(array,
                            `子区间 [${p + 1}, ${high}] 已有序`,
                            [], '', { range: [p + 1, high] }));
                    }
                } else if (low >= 0 && low < array.length && low === high) {
                    array[low].state = 'sorted-quick';
                    steps.push(createSnapshot(array,
                        `单元素区间 [${low}, ${high}] 已有序`,
                        [low], '', { range: [low, high] }));
                }
            }

            quickSort(0, array.length - 1);
            array.forEach(x => x.state = 'sorted-quick');
            steps.push(createSnapshot(array, `排序完成！`));
            return steps;
        }

        // ======================== UI Controller ========================
        class Visualizer {
            constructor(type, containerId, descId, counterId, generatorFn) {
                this.type = type;
                this.container = document.getElementById(containerId);
                this.descEl = document.getElementById(descId);
                this.counterEl = document.getElementById(counterId);
                this.generatorFn = generatorFn;
                this.steps = [];
                this.currentIndex = -1;
                this.playing = false;
                this.timer = null;
                this.baseArray = [];

                const section = this.container.closest('section');
                this.inputEl = section.querySelector('.array-input');
                this.btnGenerate = section.querySelector('.btn-generate');
                this.btnReset = section.querySelector('.btn-reset');
                this.btnPrev = section.querySelector('.btn-prev');
                this.btnNext = section.querySelector('.btn-next');
                this.btnPlay = section.querySelector('.btn-play');

                this.bindEvents();
            }

            bindEvents() {
                this.btnGenerate.addEventListener('click', () => this.generate());
                this.btnReset.addEventListener('click', () => this.reset());
                this.btnPrev.addEventListener('click', () => this.prev());
                this.btnNext.addEventListener('click', () => this.next());
                this.btnPlay.addEventListener('click', () => this.togglePlay());
            }

            generate() {
                this.pause();
                let raw = parseInput(this.inputEl.value);
                if (raw.length === 0) {
                    this.descEl.textContent = '请输入有效的数字数组';
                    return;
                }
                raw = clampArray(raw, 20);
                if (raw.some(v => v < 0)) {
                    this.descEl.textContent = '请输入非负数';
                    return;
                }
                this.baseArray = [...raw];
                this.steps = this.generatorFn(raw);
                this.currentIndex = -1;
                this.renderEmpty();
                this.updateInfo();
            }

            reset() {
                this.pause();
                if (this.baseArray.length === 0) {
                    this.generate();
                    return;
                }
                this.steps = this.generatorFn(this.baseArray);
                this.currentIndex = -1;
                this.renderEmpty();
                this.updateInfo();
            }

            createCardWrapper(id, val) {
                const wrap = document.createElement('div');
                wrap.className = 'card-wrapper';
                wrap.dataset.id = id;

                const card = document.createElement('div');
                card.className = 'card';
                const maxVal = Math.max(...this.baseArray, 1);
                const unitH = 200 / maxVal;
                const h = Math.max(28, Math.round(val * unitH));
                card.style.height = `${h}px`;

                if (this.type === 'selection') {
                    const crown = document.createElement('div');
                    crown.className = 'crown';
                    crown.textContent = 'MIN';
                    card.appendChild(crown);
                } else if (this.type === 'quick') {
                    const badges = document.createElement('div');
                    badges.className = 'card-badges';
                    const badgeI = document.createElement('span');
                    badgeI.className = 'badge-i';
                    badgeI.textContent = 'i';
                    const badgeJ = document.createElement('span');
                    badgeJ.className = 'badge-j';
                    badgeJ.textContent = 'j';
                    badges.appendChild(badgeI);
                    badges.appendChild(badgeJ);
                    card.appendChild(badges);
                }

                const valSpan = document.createElement('span');
                valSpan.className = 'val';
                valSpan.textContent = val;

                card.appendChild(valSpan);
                wrap.appendChild(card);

                return wrap;
            }

            renderEmpty() {
                this.container.innerHTML = '';
                this.baseArray.forEach((val, idx) => {
                    this.container.appendChild(this.createCardWrapper(idx, val));
                });
                const cardWidth = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--card-width')) || 44;
                const cardGap = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--card-gap')) || 8;
                const stepSize = cardWidth + cardGap;
                const wrappers = Array.from(this.container.querySelectorAll('.card-wrapper'));
                const baseOffset = wrappers[0] ? wrappers[0].offsetLeft : 0;
                this.baseArray.forEach((_, idx) => {
                    const idxDiv = document.createElement('div');
                    idxDiv.className = 'card-index-overlay';
                    idxDiv.textContent = idx;
                    idxDiv.style.left = `${baseOffset + idx * stepSize}px`;
                    this.container.appendChild(idxDiv);
                });
            }

            renderStep(index) {
                if (index < 0 || index >= this.steps.length) return;
                const step = this.steps[index];
                const maxVal = Math.max(...this.baseArray, 1);
                const unitH = 200 / maxVal;
                const cardWidth = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--card-width')) || 44;
                const cardGap = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--card-gap')) || 8;
                const stepSize = cardWidth + cardGap;

                // Ensure wrappers exist keyed by id
                let wrappers = Array.from(this.container.querySelectorAll('.card-wrapper'));
                const wrapperById = new Map();
                wrappers.forEach(w => wrapperById.set(Number(w.dataset.id), w));

                step.array.forEach(item => {
                    if (!wrapperById.has(item.id)) {
                        const wrap = this.createCardWrapper(item.id, item.value);
                        wrapperById.set(item.id, wrap);
                        this.container.appendChild(wrap);
                    }
                });

                // Reset all
                wrapperById.forEach(w => {
                    w.className = 'card-wrapper';
                    w.style.transform = '';
                    w.style.zIndex = '';
                    const card = w.querySelector('.card');
                    if (card) card.style.removeProperty('--swap-dir');
                });

                // Clean up old overlays
                this.container.querySelectorAll('.range-box, .vs-badge-overlay, .slot-placeholder-overlay, .card-index-overlay').forEach(el => el.remove());

                // Get base offset for absolute positioning
                wrappers = Array.from(this.container.querySelectorAll('.card-wrapper'));
                const baseOffset = wrappers[0] ? wrappers[0].offsetLeft : 0;

                // Position and style each card
                step.array.forEach((item, posIndex) => {
                    const w = wrapperById.get(item.id);
                    if (!w) return;
                    const dx = (posIndex - item.id) * stepSize;
                    w.style.transform = `translateX(${dx}px)`;

                    if (item.state && item.state !== 'default') {
                        w.classList.add(item.state);
                    }

                    const card = w.querySelector('.card');
                    const h = Math.max(28, Math.round(item.value * unitH));
                    card.style.height = `${h}px`;
                    card.querySelector('.val').textContent = item.value;

                    if (step.focus && step.focus.includes(posIndex)) {
                        w.classList.add('focused');
                        w.style.zIndex = '100';
                    }
                });

                // Apply animations to target cards only
                if (step.animation) {
                    const applyAnimToPos = (posIdx, animClass) => {
                        const item = step.array[posIdx];
                        if (!item) return;
                        const w = wrapperById.get(item.id);
                        if (w) w.classList.add(animClass);
                    };

                    if (step.animation === 'compare') {
                        step.focus.forEach(idx => applyAnimToPos(idx, 'comparing'));
                        if (this.type === 'bubble' && step.focus.length === 2) {
                            const leftPos = Math.min(step.focus[0], step.focus[1]);
                            const vs = document.createElement('div');
                            vs.className = 'vs-badge-overlay';
                            vs.textContent = 'VS';
                            vs.style.left = `${baseOffset + leftPos * stepSize + stepSize / 2}px`;
                            this.container.appendChild(vs);
                        }
                    } else if (step.animation === 'swap') {
                        if (this.type === 'bubble') {
                            step.focus.forEach((idx, i) => {
                                const item = step.array[idx];
                                if (!item) return;
                                const w = wrapperById.get(item.id);
                                if (w) {
                                    w.classList.add('anim-swap-cross');
                                    const card = w.querySelector('.card');
                                    if (card) {
                                        card.style.setProperty('--swap-dir', i === 0 ? '18px' : '-18px');
                                    }
                                }
                            });
                        } else {
                            step.focus.forEach(idx => applyAnimToPos(idx, 'swapping'));
                        }
                    } else if (step.animation === 'shift') {
                        if (this.type === 'insertion' && step.focus.length >= 2) {
                            applyAnimToPos(step.focus[1], 'shift-right');
                        }
                        if (step.extras && step.extras.slotIndex !== undefined) {
                            const slot = document.createElement('div');
                            slot.className = 'slot-placeholder-overlay';
                            slot.style.left = `${baseOffset + step.extras.slotIndex * stepSize}px`;
                            this.container.appendChild(slot);
                        }
                    } else if (step.animation === 'drop') {
                        step.focus.forEach(idx => applyAnimToPos(idx, 'anim-drop'));
                    } else if (step.animation === 'pivot-place') {
                        step.focus.forEach(idx => applyAnimToPos(idx, 'anim-pivot-place'));
                    } else if (step.animation === 'crown-transfer') {
                        if (step.extras) {
                            if (step.extras.oldMinIdx !== undefined) applyAnimToPos(step.extras.oldMinIdx, 'anim-crown-transfer');
                            if (step.extras.newMinIdx !== undefined) applyAnimToPos(step.extras.newMinIdx, 'crown-pop');
                        }
                    } else if (step.animation === 'arc-fly') {
                        step.focus.forEach(idx => applyAnimToPos(idx, 'anim-arc-fly'));
                    }
                }

                // Selection sort crowns
                if (this.type === 'selection') {
                    let effectiveMinIdx = step.extras ? step.extras.minIdx : undefined;
                    if (step.animation === 'crown-transfer' && step.extras && step.extras.newMinIdx !== undefined) {
                        effectiveMinIdx = step.extras.newMinIdx;
                    }
                    if (effectiveMinIdx !== undefined) {
                        const item = step.array[effectiveMinIdx];
                        if (item) {
                            const w = wrapperById.get(item.id);
                            if (w) w.classList.add('has-crown', 'min-current');
                        }
                    }
                }

                // Insertion sort slot when not in shift animation
                if (this.type === 'insertion' && step.extras && step.extras.slotIndex !== undefined && step.animation !== 'shift') {
                    const slot = document.createElement('div');
                    slot.className = 'slot-placeholder-overlay';
                    slot.style.left = `${baseOffset + step.extras.slotIndex * stepSize}px`;
                    this.container.appendChild(slot);
                }

                // Quick sort badges
                if (this.type === 'quick') {
                    if (step.extras && step.extras.i !== undefined && step.extras.i >= 0) {
                        const item = step.array[step.extras.i];
                        if (item) {
                            const w = wrapperById.get(item.id);
                            if (w) w.classList.add('show-badge-i');
                        }
                    }
                    if (step.extras && step.extras.j !== undefined) {
                        const item = step.array[step.extras.j];
                        if (item) {
                            const w = wrapperById.get(item.id);
                            if (w) w.classList.add('show-badge-j');
                        }
                    }
                }

                // Range boxes for quick sort
                if (this.type === 'quick' && step.extras && step.extras.range) {
                    const [low, high] = step.extras.range;
                    const box = document.createElement('div');
                    box.className = 'range-box';
                    const leftX = baseOffset + low * stepSize - 4;
                    const width = (high - low + 1) * stepSize + 4;
                    box.style.left = `${leftX}px`;
                    box.style.width = `${width}px`;
                    this.container.appendChild(box);
                }

                // Card index overlays
                step.array.forEach((_, posIndex) => {
                    const idxDiv = document.createElement('div');
                    idxDiv.className = 'card-index-overlay';
                    idxDiv.textContent = posIndex;
                    idxDiv.style.left = `${baseOffset + posIndex * stepSize}px`;
                    this.container.appendChild(idxDiv);
                });

                this.descEl.textContent = step.desc;
                this.counterEl.textContent = `步骤 ${index + 1} / ${this.steps.length}`;

                // Animation classes are reset at the top of renderStep via w.className = 'card-wrapper'
            }

            updateInfo() {
                if (this.currentIndex === -1) {
                    this.descEl.textContent = '请点击“下一步”开始演示';
                    this.counterEl.textContent = `- / ${this.steps.length}`;
                } else {
                    this.renderStep(this.currentIndex);
                }
            }

            next() {
                if (this.steps.length === 0) {
                    this.generate();
                    return;
                }
                if (this.currentIndex < this.steps.length - 1) {
                    this.currentIndex++;
                    this.renderStep(this.currentIndex);
                } else {
                    this.pause();
                }
            }

            prev() {
                if (this.steps.length === 0) return;
                if (this.currentIndex >= 0) {
                    this.currentIndex--;
                    if (this.currentIndex === -1) {
                        this.renderEmpty();
                        this.updateInfo();
                    } else {
                        this.renderStep(this.currentIndex);
                    }
                }
            }

            togglePlay() {
                if (this.playing) {
                    this.pause();
                } else {
                    this.play();
                }
            }

            play() {
                if (this.steps.length === 0) {
                    this.generate();
                    return;
                }
                this.playing = true;
                this.btnPlay.textContent = '暂停';
                if (this.currentIndex >= this.steps.length - 1) {
                    this.currentIndex = -1;
                }
                this.timer = setInterval(() => {
                    if (this.currentIndex < this.steps.length - 1) {
                        this.currentIndex++;
                        this.renderStep(this.currentIndex);
                    } else {
                        this.pause();
                    }
                }, 1100);
            }

            pause() {
                this.playing = false;
                this.btnPlay.textContent = '自动播放';
                if (this.timer) {
                    clearInterval(this.timer);
                    this.timer = null;
                }
            }
        }

        // ======================== Initialize ========================
        const visualizers = {};

        document.addEventListener('DOMContentLoaded', () => {
            visualizers.bubble = new Visualizer('bubble', 'viz-bubble', 'desc-bubble', 'counter-bubble', generateBubbleSteps);
            visualizers.selection = new Visualizer('selection', 'viz-selection', 'desc-selection', 'counter-selection', generateSelectionSteps);
            visualizers.insertion = new Visualizer('insertion', 'viz-insertion', 'desc-insertion', 'counter-insertion', generateInsertionSteps);
            visualizers.quick = new Visualizer('quick', 'viz-quick', 'desc-quick', 'counter-quick', generateQuickSteps);

            Reveal.on('slidechanged', () => {
                Object.values(visualizers).forEach(v => v.pause());
            });
        });
    </script>
